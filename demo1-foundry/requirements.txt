# Demo 1 — Foundry A2A agent (Microsoft Docs Expert)
#
# Install in a Demo-1-only venv:
#   python -m venv .venv-demo1
#   .\.venv-demo1\Scripts\Activate.ps1
#   pip install --pre -r demo1-foundry\requirements.txt
#
# Why a separate venv: Demo 1 talks to Foundry's A2A endpoint with
# a2a-sdk 1.0.x (protobuf bindings). Demo 2's agent-framework-a2a beta
# requires a2a-sdk <0.3.24. They cannot coexist in a single venv.

# ─── Common ─────────────────────────────────────────────────────────────────
python-dotenv>=1.0.1
httpx>=0.28.1
azure-identity>=1.25.3

# ─── Foundry SDK (preview v2.x — has A2A + MCP) ─────────────────────────────
# Stable v2.0+ ships A2A enablement on agents and the MCP tool.
azure-ai-projects>=2.1.0,<3

# ─── A2A protocol SDK 1.0.x (protobuf / gRPC bindings) ──────────────────────
# Foundry's A2A endpoint exposes protocol v0.3 over JSON-RPC + the gRPC
# protobuf surface. a2a-sdk 1.0.x is the maintained client SDK for this.
a2a-sdk==1.0.2
protobuf>=5.29.5,<6
