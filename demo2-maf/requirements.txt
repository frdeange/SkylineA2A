# Demo 2 — Microsoft Agent Framework A2A server (DevOps Helper)
#
# Install in a Demo-2-only venv:
#   python -m venv .venv-demo2
#   .\.venv-demo2\Scripts\Activate.ps1
#   pip install --pre -r demo2-maf\requirements.txt
#
# Why install agent-framework-a2a from a git commit (not PyPI):
#   The PyPI wheel `agent-framework-a2a==1.0.0b260507` still pins
#   a2a-sdk<0.3.24 (the old API). The GitHub `main` branch migrated to
#   a2a-sdk>=1.0.0 in commit 4ad96b6 (PR #5752 — "[BREAKING] Migrate
#   agent-framework-a2a to a2a-sdk v1.0"). The canonical sample uses the
#   new API, so we pin to a SHA on `main` that includes that migration.
#   This way the demo is reproducible: the SHA is frozen even if `main`
#   moves on.
#
# Why a separate venv from Demo 1:
#   Both demos now use a2a-sdk 1.0.x and can technically share a venv,
#   but Demo 2 pulls in the heavy agent-framework-* chain. Keeping them
#   isolated is cleaner for the walkthrough.

# ─── Common ─────────────────────────────────────────────────────────────────
python-dotenv>=1.0.1
httpx>=0.28.1
azure-identity>=1.25.3

# ─── Microsoft Agent Framework (stable 1.3.x from PyPI) ─────────────────────
# core = base SDK (Agent, ChatClient protocol, function-tool decorators).
# foundry = FoundryChatClient — uses Foundry-deployed models from MAF.
agent-framework-core>=1.3.0,<2
agent-framework-foundry>=1.3.0,<2

# ─── MAF A2A bridge from GitHub `main` (commit cfd3dfe, includes #5752) ─────
# This is the v1.0-API version of the package. PyPI's beta is still on 0.3.x.
agent-framework-a2a @ git+https://github.com/microsoft/agent-framework@cfd3dfe40b6c7bee8bfcb3b02de6cc88e4522784#subdirectory=python/packages/a2a

# ─── A2A 1.0.x SDK with the http-server extras (Starlette routes, SSE) ──────
# Same a2a-sdk major version as Demo 1. The http-server extra adds
# starlette + sse-starlette + fastapi which a2a.server.routes uses.
a2a-sdk[http-server]>=1.0.0,<2

# ─── Demo 2 server runtime ──────────────────────────────────────────────────
uvicorn[standard]>=0.32.0
starlette>=0.41.0

# ─── Async HTTP transport ───────────────────────────────────────────────────
# FoundryChatClient builds an async AIProjectClient, whose AsyncPipelineClient
# lazy-imports aiohttp at construction time. aiohttp is not pulled in by any
# of the other Azure SDK packages, so we declare it explicitly here.
aiohttp>=3.10.0

# ─── Protobuf compatibility cap ─────────────────────────────────────────────
# a2a-sdk 1.0.x declares `protobuf>=5.29.5` with no upper bound. With pip's
# `--pre` flag (needed for agent-framework-a2a's pre-release), protobuf
# resolves to 7.x release candidates, where `FieldDescriptor.label` was
# removed — breaking a2a-sdk's protobuf-based message handlers with:
#     'google._upb._message.FieldDescriptor' object has no attribute 'label'
# Stable 5.x/6.x both work; cap below 7 until a2a-sdk validates against it.
protobuf>=5.29.5,<7
