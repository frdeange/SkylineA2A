# Demo 2 — Microsoft Agent Framework A2A server (DevOps Helper)
#
# Install in a Demo-2-only venv:
#   python -m venv .venv-demo2
#   .\.venv-demo2\Scripts\Activate.ps1
#   pip install --pre -r demo2-maf\requirements.txt
#
# Why a separate venv: agent-framework-a2a is currently a beta that pins
# a2a-sdk>=0.3.5,<0.3.24 — incompatible with Demo 1's a2a-sdk 1.0.x.
# --pre is required because agent-framework-a2a is a pre-release.

# ─── Common ─────────────────────────────────────────────────────────────────
python-dotenv>=1.0.1
httpx>=0.28.1
azure-identity>=1.25.3

# ─── Microsoft Agent Framework (stable 1.3.x) ───────────────────────────────
# core = base SDK (Agent, ChatClient protocol, function-tool decorators).
# foundry = FoundryChatClient — uses Foundry-deployed models from MAF.
agent-framework-core>=1.3.0,<2
agent-framework-foundry>=1.3.0,<2

# ─── MAF A2A bridge (beta) + transitive a2a-sdk 0.3.x ───────────────────────
# A2AExecutor adapter; the package pins a2a-sdk<0.3.24 transitively.
agent-framework-a2a==1.0.0b260507

# ─── A2A 0.3.x SDK with the http-server extras (Starlette routes, SSE) ──────
# Pinned matching agent-framework-a2a's constraint. The "http-server" extra
# brings in starlette + sse-starlette + fastapi which a2a.server.routes uses.
a2a-sdk[http-server]>=0.3.5,<0.3.24

# ─── Demo 2 server runtime ──────────────────────────────────────────────────
uvicorn[standard]>=0.32.0
starlette>=0.41.0
